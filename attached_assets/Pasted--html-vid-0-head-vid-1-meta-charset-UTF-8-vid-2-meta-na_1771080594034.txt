    <script vid="9">
        const container = document.getElementById('visual-container');
        const canvas = document.createElement('canvas');
        container.appendChild(canvas);
        const ctx = canvas.getContext('2d');
        
        let width, height;
        let animationFrameId;

        
        const config = {
            count: 800,           
            baseSize: 2.5,        
            spread: 12,            
            speed: 0.05,          
            colorShiftSpeed: 0.2  
        };

        
        function resize() {
            width = container.clientWidth;
            height = container.clientHeight;

            const dpr = window.devicePixelRatio || 1;
            canvas.width = width * dpr;
            canvas.height = height * dpr;
            ctx.scale(dpr, dpr);

            canvas.style.width = width + 'px';
            canvas.style.height = height + 'px';
        }

        let tick = 0;
        let mouseX = 0;
        let mouseY = 0;
        let targetRotateX = 0;
        let targetRotateY = 0;
        let currentRotateX = 0;
        let currentRotateY = 0;

        function animate() {
            
            ctx.clearRect(0, 0, width, height);

            const centerX = width / 2;
            const centerY = height / 2;

            
            currentRotateX += (targetRotateX - currentRotateX) * 0.05;
            currentRotateY += (targetRotateY - currentRotateY) * 0.05;

            
            const gradient = ctx.createRadialGradient(centerX, centerY, 0, centerX, centerY, 200);
            gradient.addColorStop(0, 'rgba(109, 40, 217, 0.1)'); 
            gradient.addColorStop(1, 'rgba(0, 0, 0, 0)');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, width, height);

            
            for (let i = 0; i < config.count; i++) {
                
                const angle = i * 137.5 * (Math.PI / 180);
                
                
                const radius = config.spread * Math.sqrt(i);

                
                let x = centerX + radius * Math.cos(angle + tick * 0.001);
                let y = centerY + radius * Math.sin(angle + tick * 0.001);

                
                const distX = x - centerX;
                const distY = y - centerY;
                
                x += distY * currentRotateX * 0.0005;
                y += distX * currentRotateY * 0.0005;

                
                const breath = Math.sin(tick * 0.02) * 0.1 + 1;
                const size = (config.baseSize + (i / config.count) * 2) * breath;

                
                const hue = (40 + (i * 0.2) + tick * 0.5) % 360;
                const sat = 80;
                const light = 50 + Math.sin(tick * 0.05 + i * 0.01) * 20;
                
                
                const alpha = Math.max(0.1, 1 - (radius / (Math.min(width, height) * 0.6)));

                if (alpha > 0.01) {
                    ctx.beginPath();
                    ctx.arc(x, y, size, 0, Math.PI * 2);
                    ctx.fillStyle = `hsla(${hue}, ${sat}%, ${light}%, ${alpha})`;

                    
                    if (i % 15 === 0) {
                        ctx.shadowBlur = 10;
                        ctx.shadowColor = `hsla(${hue}, ${sat}%, ${light}%, ${alpha})`;
                    } else {
                        ctx.shadowBlur = 0;
                    }

                    ctx.fill();
                }
            }

            tick += 1;
            animationFrameId = requestAnimationFrame(animate);
        }

        
        resize();
        window.addEventListener('resize', resize);
        animate();

        
        document.addEventListener('mousemove', (e) => {
            const x = e.clientX - window.innerWidth / 2;
            const y = e.clientY - window.innerHeight / 2;
            targetRotateX = y;
            targetRotateY = x;
        });

        
        document.addEventListener('touchmove', (e) => {
            const x = e.touches[0].clientX - window.innerWidth / 2;
            const y = e.touches[0].clientY - window.innerHeight / 2;
            targetRotateX = y;
            targetRotateY = x;
        });

        
        if (window.DeviceOrientationEvent) {
            window.addEventListener("deviceorientation", function(event) {
                if(event.beta && event.gamma) {
                    targetRotateX = (event.beta - 45) * 5; 
                    targetRotateY = event.gamma * 5;
                }
            }, true);
        }
    </script>
